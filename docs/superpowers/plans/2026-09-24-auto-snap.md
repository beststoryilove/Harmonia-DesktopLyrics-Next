```js
/**
 * 判断句点是否应并入当前拉丁词（留在**词尾**）。
 *
 * 为什么句点要单独放宽
 * ──────────────────
 * `isLatinWordJoiner` 要求连接符**前后**都是字母/数字，但 `Mr.` 的句点
 * 后面是空格，于是被拆成独立标点，得到 `['Mr', '.', 'Smith']` ——
 * 与「句点缩写应留在词内」的要求矛盾。
 *
 * 约束：只在**词尾**吸收，且只吸收一个。
 *  · `Mr.`     → 并入 → `Mr.`
 *  · `U.S.A.`  → 词中句点由 isLatinWordJoiner 处理，词尾这个由此函数吸收
 *  · `Wait...` → 只并一个 → `Wait.` + `.` + `.`（省略号不整体吞并，
 *                否则单元宽度会异常膨胀，且列宽按最长单元计算会被带偏）
 *
 * @param {string} buffer 当前累积的词
 * @returns {boolean} 是否应吸收该句点
 */
export function shouldAbsorbPeriod(buffer) {
  if (!buffer) return false;
  if (buffer.endsWith('.')) return false; // 已吸收过，避免吞并省略号
  return isLatinLetter(buffer[0]) || isDigit(buffer[0]);
}
```

然后把 `splitVerticalUnits` 循环里处理 `isLatinWordJoiner(ch)` 的分支改成：

```js

    if (isLatinWordJoiner(ch)) {
      // 句点单独放宽：允许留在词尾（`Mr.`），但只吸收一个（`...` 不吞并）
```

```js
/**
 * 默认解除吸附的触发距离（像素）。
 *
 * 为什么需要两个阈值（迟滞 / hysteresis）
 * ────────────────────────────────────
 * 若吸附与解除共用 60px，窗口停在 60px 附近时会出现：
 *   吸附 → 位置变化使距离 > 60 → 解除 → 距离 < 60 → 再吸附 → …
 * 表现为窗口在边缘反复抖动。
 *
 * 96 = 60 + 36，迟滞带宽为吸附阈值的 60%，足以吸收鼠标与缩放的抖动。
 */
export const DEFAULT_UNSNAP_THRESHOLD = 96;

/**
 * 判断已吸附的窗口是否应解除吸附。
 *
 * @param {{x: number, y: number, width: number, height: number}} bounds 窗口矩形
 * @param {{x: number, y: number, width: number, height: number}} workArea 工作区矩形
 * @param {string} edge 当前吸附的边
 * @param {number} [threshold=DEFAULT_UNSNAP_THRESHOLD] 解除距离（像素）
 * @returns {boolean} 是否解除
 */
export function decideUnsnap(bounds, workArea, edge, threshold = DEFAULT_UNSNAP_THRESHOLD) {
  // 无法判定时按「解除」处理：让窗口回到用户可控的普通状态，
  // 好过卡在一个来路不明的吸附态里。
  if (!bounds || !workArea) return true;
  if (!SNAP_EDGES.includes(edge)) return true;

  const limit = Number.isFinite(threshold) ? Math.max(0, threshold) : DEFAULT_UNSNAP_THRESHOLD;
  const distance = edgeDistances(bounds, workArea)[edge];
  if (!Number.isFinite(distance)) return true;
  return distance > limit;
}
```

```js
/**
 * 竖排布局核心（纯函数，无 DOM / Electron 依赖，便于单元测试）。
 *
 * 职责
 * ────
 *  1. **单元 ↔ 时间映射**：把「竖排单元」对齐到 TTML 的逐字时序；
 *  2. **列度量**：算出一列需要多宽 / 多高；
 *  3. **整窗宽度预算**：按整首歌算出稳定的窗口宽度；
 *  4. **滚动偏移**：让当前唱到的单元保持在视口内。
 *
 * 为什么需要「单元 ↔ 时间映射」
 * ──────────────────────────
 * TTML 的切分粒度**比单词还细** —— 它是为逐字时序服务的音节切分。
 * 真实样本（samples/encanto-bruno.ttml）：
 *
 *   text  = "We don't talk about Bruno, no, no, no"
 *   words = ["We"," don't"," talk"," about"," Bru","no,", …]
 *                                              └────┬────┘
 *                                          Bruno 被切成两段
 *
 * 竖排若直接按 words 排列，`Bruno` 会被拆成两行且中间断字。
 * 因此竖排以 `line.text` 为切分输入（按语种边界重新切分 →
 * `Bruno` 合回整词、`,` 独立成单元），再用下面的累积偏移把时间接回来。
 *
 * 累积偏移为什么成立
 * ────────────────
 * `splitVerticalUnits` 会**丢弃所有空白**，因此恒成立：
 *
 *   各单元文本拼接 === 源文本去掉全部空白码点后的序列
 *
 * 于是只要维护一张「非空白码点序号 → 时间区间」表，
 * 就能用「前面已消耗的码点数」把每个单元精确对齐上去，
 * 无需给单元附加偏移字段（`vertical-text.js` 也无需改动接口）。
 */

import { splitVerticalUnits } from './vertical-text.js';

/** 竖排默认度量。字号与横排的 CSS 保持一致（见 lyrics.css 的注释）。 */
export const DEFAULT_VERTICAL_METRICS = Object.freeze({
  /** 主列字号（与横排 .main-inner 的 26px 对齐） */
  cjkSize: 26,
  latinSize: 22,
  /** 副列字号（对唱 / 背景行，弱化） */
  bgCjkSize: 16,
  bgLatinSize: 14,
  /** 翻译列字号 */
  transSize: 13,
  /** 单元间距 */
  gap: 2,
  /** 列左右内边距合计 */
  paddingX: 24,
  /** 列间距 */
  columnGap: 10,
  /** 单元格平均宽度 / 字号 的比值（拉丁字符估算用） */
  avgLatinCharRatio: 0.58,
  /** 窗口宽度夹取范围 */
  minWidth: 120,
  maxWidth: 320,
});

/**
 * 把一行歌词映射为带时间的竖排单元。
 *
 * @param {{text: string, words: Array<{word: string, startTime: number, endTime: number}>}} line 行数据
 * @returns {Array<{text: string, kind: string, startTime: number, endTime: number}>} 带时间的单元
 */
export function buildTimedUnits(line) {
  const text = line && typeof line.text === 'string' ? line.text : '';
  const words = line && Array.isArray(line.words) ? line.words : [];
  if (!text || !words.length) return [];

  // ── 1) 建立「非空白码点序号 → [start, end]」表 ──
  //
  // 一个词的时间区间按其**码点**数分摊到各字符上，使逐字填充连续。
  // 空白码点不占时间轴（竖排不显示空格，也没有时间意义）。
  /** @type {Array<[number, number]>} */
  const times = [];
  for (const word of words) {
    const chars = Array.from(word.word || '');
    if (!chars.length) continue;
    const start = Number(word.startTime) || 0;
    const end = Number(word.endTime) || start + 1;
    const span = Math.max(1, chars.length);
    for (let i = 0; i < chars.length; i += 1) {
      if (/\s/.test(chars[i])) continue;
      times.push([
        start + ((end - start) * i) / span,
        start + ((end - start) * (i + 1)) / span,
      ]);
    }
  }
  if (!times.length) return [];

  // ── 2) 按累积偏移把单元对齐到时间表 ──
  const out = [];
  let offset = 0;
  for (const unit of splitVerticalUnits(text)) {
    const length = Array.from(unit.text).length;
    if (offset >= times.length) break; // 词数不足以覆盖文本：截断而非抛错
    const last = Math.min(offset + length - 1, times.length - 1);
    out.push({
      text: unit.text,
      kind: unit.kind,
      startTime: times[offset][0],
      endTime: times[last][1],
    });
    offset += length;
  }
  return out;
}

/**
 * 度量一列所需尺寸。
 *
 * @param {Array<{text: string, kind: string}>} units 单元数组
 * @param {object} metrics 度量参数（见 DEFAULT_VERTICAL_METRICS）
 * @param {number} [metrics.cjkSize] CJK 单元字号
 * @param {number} [metrics.latinSize] 拉丁单元字号
 * @param {number} [metrics.gap] 单元间距
 * @param {number} [metrics.paddingX] 左右内边距合计
 * @param {number} [metrics.avgLatinCharRatio] 拉丁字符平均宽 / 字号
 * @returns {{width: number, height: number}} 列尺寸
 */
export function measureColumn(units, metrics = DEFAULT_VERTICAL_METRICS) {
  const cjkSize = Number(metrics.cjkSize) || DEFAULT_VERTICAL_METRICS.cjkSize;
  const latinSize = Number(metrics.latinSize) || DEFAULT_VERTICAL_METRICS.latinSize;
  const gap = Number.isFinite(metrics.gap) ? metrics.gap : DEFAULT_VERTICAL_METRICS.gap;
  const paddingX = Number.isFinite(metrics.paddingX) ? metrics.paddingX : DEFAULT_VERTICAL_METRICS.paddingX;
  const ratio = Number.isFinite(metrics.avgLatinCharRatio)
    ? metrics.avgLatinCharRatio
    : DEFAULT_VERTICAL_METRICS.avgLatinCharRatio;

  if (!Array.isArray(units) || !units.length) return { width: Math.round(paddingX), height: 0 };

  let height = 0;
  let widest = cjkSize;
  for (const unit of units) {
    const isCjk = unit.kind === 'cjk';
    height += isCjk ? cjkSize : latinSize;
    height += gap;
    const width = isCjk ? cjkSize : Math.round(unit.text.length * latinSize * ratio);
    if (width > widest) widest = width;
  }
  if (height > 0) height -= gap; // 末尾不留间距

  return { width: Math.round(widest + paddingX), height: Math.round(height) };
}

/** 把数值限制在区间内。 */
function clamp(value, min, max) {
  if (max < min) return min;
  return Math.min(Math.max(value, min), max);
}

/**
 * 按**整首歌**预算竖排窗口宽度。
 *
 * 为什么必须看整首歌而不是当前句
 * ────────────────────────────
 * 若宽度按当前行自适应，英文歌词每 3–4 秒换一句、句长差异很大，会导致：
 *   ① 窗口宽度持续伸缩（视觉抖动）；
 *   ② 窗口伸缩触发 moved → 再次进入吸附判定 → **反馈环**。
 * 因此宽度必须一次算定、整个会话内保持稳定。
 *
 * @param {Array<object>} lines 全部歌词行（session.lines），每项含 text / words / isBG / translatedLyric
 * @param {object} [metrics] 度量参数
 * @returns {{width: number, mainWidth: number, bgWidth: number, transWidth: number, hasBg: boolean, hasTranslation: boolean}} 布局结果
 */
export function computeVerticalLayout(lines, metrics = DEFAULT_VERTICAL_METRICS) {
  const m = { ...DEFAULT_VERTICAL_METRICS, ...metrics };
  const list = Array.isArray(lines) ? lines : [];

  let mainWidth = 0;
  let bgWidth = 0;
  let transWidth = 0;
  let hasBg = false;
  let hasTranslation = false;

  for (const line of list) {
    if (!line) continue;
    const units = buildTimedUnits(line);
    if (!units.length) continue;

    if (line.isBG) {
      hasBg = true;
      const { width } = measureColumn(units, { ...m, cjkSize: m.bgCjkSize, latinSize: m.bgLatinSize });
      if (width > bgWidth) bgWidth = width;
    } else {
      const { width } = measureColumn(units, m);
      if (width > mainWidth) mainWidth = width;
    }

    // 翻译按纯文本估算列宽（它没有逐字时序，不需要 buildTimedUnits）
    const translation = line.translatedLyric || '';
    if (translation) {
      hasTranslation = true;
      const width = Math.round(
        Array.from(translation).length * m.transSize * m.avgLatinCharRatio + m.paddingX,
      );
      if (width > transWidth) transWidth = width;
    }
  }

  // 一列都没有时（未连接 / 歌词为空）回落到最小宽度
  if (!mainWidth && !bgWidth && !transWidth) {
    return { width: m.minWidth, mainWidth: 0, bgWidth: 0, transWidth: 0, hasBg: false, hasTranslation: false };
  }

  const columnCount = [mainWidth, hasBg ? bgWidth : 0, hasTranslation ? transWidth : 0]
    .filter((value) => value > 0).length;
  const gaps = Math.max(0, columnCount - 1) * m.columnGap;
  const total = mainWidth + (hasBg ? bgWidth : 0) + (hasTranslation ? transWidth : 0) + gaps;

  return {
    width: clamp(Math.round(total), m.minWidth, m.maxWidth),
    mainWidth,
    bgWidth: hasBg ? bgWidth : 0,
    transWidth: hasTranslation ? transWidth : 0,
    hasBg,
    hasTranslation,
  };
}

/**
 * 计算竖排列的纵向滚动偏移，使活跃单元保持在视口内。
 *
 * 与横排 `KaraokeLine._updateScroll` 同策略（活跃单元居中），只是把 X 换成 Y。
 *
 * @param {object} input 输入
 * @param {number} input.totalSize 内容总高
 * @param {number} input.viewSize 视口高度
 * @param {number} input.activeStart 活跃单元相对内容顶部的起点
 * @param {number} input.activeEnd 活跃单元相对内容顶部的终点
 * @returns {number} 应施加的位移（非负，配合 translateY(-offset)）
 */
export function computeScrollOffset(input) {
  if (!input) return 0;
  const totalSize = Number(input.totalSize);
  const viewSize = Number(input.viewSize);
  const activeStart = Number(input.activeStart) || 0;
  const activeEnd = Number(input.activeEnd) || activeStart;

  if (!Number.isFinite(totalSize) || !Number.isFinite(viewSize)) return 0;
  if (viewSize <= 0 || totalSize <= viewSize) return 0;

  const maxScroll = totalSize - viewSize;
  const target = (activeStart + activeEnd) / 2 - viewSize / 2;
  return clamp(Math.round(target), 0, Math.round(maxScroll));
}
```

```js
/**
 * 设置持久化（纯 Node，无 Electron 依赖，便于单元测试）。
 *
 * 设计原则
 * ────────
 *  · **损坏的配置绝不能导致启动失败** —— 任何异常都回落默认值；
 *  · **原子写** —— 先写 `.tmp` 再 rename，避免断电/崩溃留下半截 JSON；
 *  · **只存已知字段** —— 读入时收敛，丢弃未知键（避免配置文件被塞入任意内容）。
 *
 * 目前只有「自动吸附开关」一项。刻意不预埋其他配置：需要时再加，
 * 避免出现一堆从未被读取的字段。
 */

import { readFileSync, writeFileSync, renameSync, mkdirSync, unlinkSync } from 'node:fs';
import { dirname } from 'node:path';

/** 默认设置。 */
export const DEFAULT_SETTINGS = Object.freeze({
  /** 自动吸附开关，默认**关闭** */
  snapEnabled: false,
});

/**
 * 把任意输入收敛为合法设置。
 *
 * @param {unknown} raw 原始数据
 * @returns {{snapEnabled: boolean}} 合法设置
 */
export function normalizeSettings(raw) {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
    return { ...DEFAULT_SETTINGS };
  }
  return {
    // 用 Boolean() 而非 === true：配置文件里出现 "yes" / 1 这类真值也认，
    // 但 undefined / null / 0 / '' 一律为 false。
    snapEnabled: Boolean(raw.snapEnabled),
  };
}

/**
 * 读取设置文件。任何失败都返回默认值。
 *
 * @param {string} filePath 设置文件路径
 * @returns {{snapEnabled: boolean}} 设置
 */
export function readSettings(filePath) {
  try {
    const text = readFileSync(filePath, 'utf8');
    return normalizeSettings(JSON.parse(text));
  } catch (_) {
    // 文件不存在 / 无权限 / JSON 损坏 —— 全部回落默认值，绝不抛给调用方
    return { ...DEFAULT_SETTINGS };
  }
}

/**
 * 原子写入设置文件。
 *
 * @param {string} filePath 设置文件路径
 * @param {object} settings 设置（会被 normalize）
 * @returns {boolean} 是否写入成功
 */
export function writeSettings(filePath, settings) {
  const payload = normalizeSettings(settings);
  const tempPath = `${filePath}.tmp`;
  try {
    mkdirSync(dirname(filePath), { recursive: true });
    writeFileSync(tempPath, `${JSON.stringify(payload, null, 2)}\n`, 'utf8');
    renameSync(tempPath, filePath);
    return true;
  } catch (_) {
    // 写失败不应影响主流程（设置丢了总比应用崩了强）
    try {
      unlinkSync(tempPath);
    } catch (_) {
      /* 清理失败忽略 */
    }
    return false;
  }
}
```

> **⚠️ 计划原文在这里设计错了，已按 Task 5 的实测更正。**
>
> 原计划在这里声明 `isDragging` / `dragTimer` 两个变量，用「`move` 之后 200ms 内
> 视为拖动中」去**近似**「拖动结束」，理由写的是「Electron 没有 dragend 事件，
> 且 `moved` 也可能由 `setBounds` 触发」。
>
> **Task 5 实测（Electron 33.4.11 / Windows）证伪了该前提**：
>
> | 操作 | 实际派发的事件 |
> |---|---|
> | `setBounds` 只改 height | `resize`（**不触发** `move` / `moved`） |
> | `setBounds` 只改 x/y | `move`（**不触发** `moved`） |
> | `setBounds` 同时改位置与尺寸 | `resize` + `move`（**不触发** `moved`） |
> | 真实 OS 鼠标拖动 | `move` ×35，末尾 `moved` ×1 |
>
> 两个结论：
> 1. **`moved` 只在真实拖动结束时派发一次** —— 它本身就是精确的「拖动结束」信号，
>    不需要被「近似」。因此这两个变量是多余的，**Task 5 已按 YAGNI 删除**。
> 2. **`move` 是同步派发的**，所以 `movingProgrammatically` 的夹心写法确实有效。
>
> ⚠️ **措辞注意**：上表是 **Windows** 上的实测结论。写注释时**不要**用平台通用的
> 口吻断言「`setBounds` 永不触发 `moved`」—— Electron 文档称在 **macOS** 上
> `moved` 是 `move` 的别名，而本项目 `main.js` 确实处理 `darwin`。
> 正确的写法是标注「实测于 Electron 33.4.11 / Windows」，并说明
> `movingProgrammatically` 守卫**无论平台如何都能打断潜在循环**。
>
> 仍然成立且必须保留：`setBounds` **会**触发 `move`，所以程序自身改尺寸时
> **必须**带 `movingProgrammatically` 守卫 —— 否则会在屏幕上闪出吸附预览条
> （`pushSnapPreview()` 挂在 `move` 上）。**这才是守卫的真实理由**，
> 不要写成「防止重放吸附判定」（那个机制不存在，会自相矛盾）。

```js

/** 取窗口当前所在的显示器工作区（多显示器下必须用**所在**显示器）。 */
function currentWorkArea(bounds) {
  const display = screen.getDisplayMatching(bounds);
  return display.workArea;
}

/**
 * 按整首歌重新预算竖排窗口宽度。
 *
 * 只在**歌词版本变化**时调用 —— 绝不能放进每 100ms 的 pushState 里：
 * 它要遍历全部歌词行做切分，且宽度必须保持稳定（见 vertical-layout.js 注释）。
 */
function syncVerticalWidth() {
  const session = sessions ? sessions.primarySession() : null;
  if (!session) {
    verticalWidth = DEFAULT_VERTICAL_METRICS.minWidth;
    return;
  }
  try {
    const layout = computeVerticalLayout(session.lines, DEFAULT_VERTICAL_METRICS);
    verticalWidth = layout.width;
  } catch (_) {
    verticalWidth = DEFAULT_VERTICAL_METRICS.minWidth;
  }
}

/** 把当前布局推给渲染层。 */
function pushLayout() {
  if (!lyricsWindow || lyricsWindow.isDestroyed()) return;
  lyricsWindow.webContents.send('hdl:layout', {
    mode: snapEdge && isVerticalEdge(snapEdge) ? 'vertical' : 'horizontal',
    edge: snapEdge,
    width: verticalWidth,
  });
}

/**
 * 执行吸附到指定边。
 *
 * @param {string} edge 目标边
 */
function applySnap(edge) {
  if (!lyricsWindow || lyricsWindow.isDestroyed()) return;

  const bounds = lyricsWindow.getBounds();
  const workArea = currentWorkArea(bounds);
  const vertical = isVerticalEdge(edge);

  // 记住吸附前的横排尺寸，供解除时还原
  if (!snapEdge) horizontalSize = { width: bounds.width, height: bounds.height };

  const size = vertical
    ? { width: Math.min(verticalWidth, workArea.width), height: workArea.height }
    : { width: Math.min(horizontalSize.width, workArea.width), height: horizontalSize.height };

  const next = computeSnappedBounds(bounds, workArea, edge, size);
  snapEdge = edge;
  movingProgrammatically = true;
  lyricsWindow.setBounds(next);
  movingProgrammatically = false;
  pushLayout();
  refreshTray();
}

/** 解除吸附，还原横排尺寸（保持在当前显示器内）。 */
function clearSnap() {
  if (!lyricsWindow || lyricsWindow.isDestroyed()) return;
  if (!snapEdge) return;

  const bounds = lyricsWindow.getBounds();
  const workArea = currentWorkArea(bounds);
  const width = Math.min(horizontalSize.width, workArea.width);
  const height = Math.min(horizontalSize.height, workArea.height);
  const x = Math.min(Math.max(bounds.x, workArea.x), workArea.x + workArea.width - width);
  const y = Math.min(Math.max(bounds.y, workArea.y), workArea.y + workArea.height - height);

  snapEdge = null;
  movingProgrammatically = true;
  lyricsWindow.setBounds({ x: Math.round(x), y: Math.round(y), width, height });
  movingProgrammatically = false;
  pushLayout();
  refreshTray();
}

/**
 * 拖动结束时的吸附判定。
 *
 * 只在**拖动结束后**判定，而不是 move 过程中：
 * `move` 在拖动中高频触发，实时吸附会在用户还在拖的时候把窗口吸走，
 * 手感割裂；且吸附改尺寸会再次触发 move，存在反馈环风险。
 */
function onMoved() {
  if (!lyricsWindow || lyricsWindow.isDestroyed()) return;

  let bounds = lyricsWindow.getBounds();
  let workArea = currentWorkArea(bounds);

  // 已吸附 → 先判断是否该解除（96px 迟滞带，避免临界抖动）
  if (snapEdge) {
    if (!decideUnsnap(bounds, workArea, snapEdge)) return; // 仍在迟滞带内，保持现状
    clearSnap();
    // ⚠️ 解除吸附会还原尺寸与位置，必须**重新读取**再判定。
    //
    // 这是一个真实踩过的坑（实施时才暴露）：竖排条是全高的（如 120×1040），
    // 其顶边与底边**同时**贴合工作区上下缘（距离均为 0）。若拿这个陈旧矩形去
    // decideSnap，它必然命中 top 或 bottom（取决于迭代顺序），与用户实际意图无关 ——
    // 表现就是「把左侧条拖到右边，却吸到了上/下边」。
    // 实测：x=1500 处陈旧矩形判为 top，还原后的 900×220 矩形才正确判为 right。
    bounds = lyricsWindow.getBounds();
    workArea = currentWorkArea(bounds);
  }

  if (!snapEnabled) return;

  const hit = decideSnap(bounds, workArea);
  if (hit) applySnap(hit.edge);
}
```

```js
/**
 * 推送吸附预览状态。
 *
 * 为什么在**窗口内缘**画预览条，而不是把窗口撑大到全屏
 * ────────────────────────────────────────────────
 * 撑大窗口会破坏 getBounds() 的真实性（/health 上报的 windowBounds 会失真），
 * 且透明窗口一旦铺满屏幕就会变成整屏拦截鼠标的事件板，
 * 用户无法再拖到其他窗口上。内缘高亮条零几何副作用、零事件副作用。
 */
function pushSnapPreview() {
  if (!lyricsWindow || lyricsWindow.isDestroyed()) return;
  let edge = null;
```

```js
/**
 * 计算竖排单元的填充比例。
 *
 * 与 `computeWordFills` 同一套数学，单独导出是为了让竖排的
 * 「向下填充」也能被 node:test 直接驱动（karaoke.js 其余部分是 DOM 相关，
 * 不便在无浏览器环境测试）。
 *
 * @param {Array<{startTime: number, endTime: number}>} units 带时间的单元
 * @param {number} timeMs 当前时间（毫秒）
 * @returns {number[]} 每个单元的填充比例（0~1）
 */
export function computeUnitFills(units, timeMs) {
  const fills = new Array(units.length);
  for (let i = 0; i < units.length; i += 1) {
    const unit = units[i];
    const start = Number(unit.startTime) || 0;
    const end = Number(unit.endTime) || start + 1;
    if (timeMs <= start) fills[i] = 0;
    else if (timeMs >= end) fills[i] = 1;
    else fills[i] = (timeMs - start) / Math.max(1, end - start);
  }
  return fills;
}
```

新增三个方法（放在 `_updateScroll` 之后）：

```js
  /**
   * 竖排渲染：每个单元一行，填充**向下**推进。
   *
   * 为什么复用 --p 变量
   * ────────────────
   * 横排的填充是「已知宽度的盒子 + overflow:hidden + ::after 按宽度裁切」。
   * 竖排只是把「宽度」换成「高度」—— 盒子的宽高都是已知的（列宽 × 单元高），
   * 因此同一套 --p 机制完全适用，无需第二套填充实现。
   */
  _renderVertical(line, timeMs, state) {
    const units = line ? buildTimedUnits(line) : [];
    const signature = units.map((u) => u.text).join('\u0001');

    if (signature !== this.signature) this._rebuildVertical(units, signature);
    if (!signature) {
      this.container.style.transform = '';
      return;
    }

    const fills = computeUnitFills(units, timeMs);
    // 逐单元写 --p：CSS 用 height: var(--p) 向下裁切
    for (let i = 0; i < this.unitElements.length; i += 1) {
      const percent = Math.max(0, Math.min(100, (fills[i] || 0) * 100));
      this.unitElements[i].style.setProperty('--p', `${percent.toFixed(2)}%`);
    }
    if (this.enableScroll) this._updateVerticalScroll(fills);
  }

  /** 重建竖排单元节点并缓存度量。 */
  _rebuildVertical(units, signature) {
    this.signature = signature;
    const fragment = document.createDocumentFragment();
    for (const unit of units) {
      const span = document.createElement('span');
      span.className = 'vertical-unit';
      span.textContent = unit.text;
      span.setAttribute('data-t', unit.text);
      span.setAttribute('data-kind', unit.kind || 'latin');
      fragment.appendChild(span);
    }
    this.container.textContent = '';
    this.container.appendChild(fragment);
    this.unitElements = Array.from(this.container.children);

    // 度量：仅在重建时做一次，帧循环内不触碰布局
    const style = getComputedStyle(this.container);
    const gap = parseFloat(style.rowGap) || 0;
    const sizes = this.unitElements.map((element) => element.getBoundingClientRect().height);
    this.unitSizes = sizes;
    this.unitOffsets = [];
    let acc = 0;
    for (const size of sizes) {
      this.unitOffsets.push(acc);
      acc += size + gap;
    }
    this.contentHeight = sizes.length ? acc - gap : 0;
    this.viewHeight = this.container.parentElement
      ? this.container.parentElement.getBoundingClientRect().height
      : 0;
    this.container.style.transform = '';
  }

  /**
   * 竖排滚动：让正在唱的单元保持在视口内。
   *
   * 这不是可选装饰 —— 真实英文歌词每行 10–15 个词，按单元排下来
   * 约 1200–1500px，而竖排窗口高约 1000px，一屏放不下。
   */
  _updateVerticalScroll(fills) {
    if (!this.unitOffsets || !this.unitOffsets.length) return;

    let activeIndex = -1;
    for (let i = 0; i < fills.length; i += 1) {
      if (fills[i] > 0 && fills[i] < 1) { activeIndex = i; break; }
    }
    if (activeIndex === -1) {
      for (let i = fills.length - 1; i >= 0; i -= 1) {
        if (fills[i] > 0) { activeIndex = i; break; }
      }
    }
    if (activeIndex === -1) activeIndex = 0;

    const offset = computeScrollOffset({
      totalSize: this.contentHeight,
      viewSize: this.viewHeight,
      activeStart: this.unitOffsets[activeIndex],
      activeEnd: this.unitOffsets[activeIndex] + (this.unitSizes[activeIndex] || 0),
    });
    this.container.style.transform = offset ? `translateY(${-offset}px)` : '';
  }
```

新增方法：

```js
  /**
   * 竖排渲染：主行 / 各副行 / 翻译各自成列，横向并排。
   *
   * 列序固定（主 / 副 / 翻译），不随吸附边翻转 —— 同一首歌在左右两侧
   * 显示一致，降低认知负担。
   */
  _renderVerticalColumns(view, timeMs, context) {
    if (!this._verticalColumns) this._buildVerticalColumns();

    const columns = this._verticalColumns;
    const bgLines = (view && view.bg) || [];
    const fg = view ? view.fg : null;

    // 每个槽位同时持有「外层列」(column，控制显示/隐藏) 与
    // 「内层容器」(inner，承载卡拉OK 节点)。
    // 两者必须分开：`KaraokeLine` 会把 transform 写在 inner 上做滚动，
    // 若把 hidden / is-on 也写在 inner 上，显示切换与滚动位移会互相干扰。
    const slots = [
      { kind: 'line', line: fg, slot: columns.main },
      { kind: 'line', line: bgLines[0] || null, slot: columns.bg[0] },
      { kind: 'line', line: bgLines[1] || null, slot: columns.bg[1] },
      { kind: 'trans', line: null, slot: columns.trans },
    ];

    for (const item of slots) {
      const { column, inner } = item.slot;

      if (item.kind === 'trans') {
        const text = fg ? (fg.translatedLyric || fg.romanLyric || '') : '';
        inner.textContent = text;
        column.hidden = !text;
        column.classList.toggle('is-on', Boolean(text));
        continue;
      }

      if (!item.line) {
        // 无内容：折叠列并按需清理卡拉OK 缓存
        if (column._karaoke) column._karaoke.clear();
        column.hidden = true;
        column.classList.remove('is-on');
        continue;
      }

      if (!column._karaoke) {
        column._karaoke = new KaraokeLine(inner, { vertical: true, enableScroll: true });
      }
      column.hidden = false;
      column._karaoke.render(item.line, timeMs, context);
      column.classList.add('is-on');
    }

    // 横排用的槽位在竖排下隐藏，避免两套 DOM 同时占位
    for (const slot of this.bgSlots) slot.el.classList.remove('is-on');
    this.mainRenderer.clear();
    if (this.mainTrans) this.mainTrans.classList.remove('is-on');
  }

  /**
   * 惰性创建竖排列容器。
   *
   * 每个槽位返回 `{column, inner}`：
   *  · `column` —— 外层 `.v-col`，负责显示/隐藏（`hidden` 属性）；
   *  · `inner`  —— 内层 `.v-col-inner`，负责承载卡拉OK 节点与滚动位移。
   * 分成两层是必需的：`KaraokeLine` 会把 `transform` 写在容器上做滚动，
   * 若显示状态与滚动位移写在同一个元素上，切换显示时会互相覆盖。
   */
  _buildVerticalColumns() {
    const root = document.createElement('div');
    root.className = 'v-columns';
    // 初始折叠；实际是否显示由 setLayout() 的 is-vertical 类控制
    root.hidden = this.layout !== 'vertical';
    const make = (className) => {
      const column = document.createElement('div');
      column.className = `v-col ${className}`;
      // 初始折叠：等有内容再展开，避免空列占位导致布局跳动
      column.hidden = true;
      const inner = document.createElement('div');
      inner.className = 'v-col-inner';
      column.appendChild(inner);
      root.appendChild(column);
      return { column, inner };
    };
    const main = make('v-col-main');
    const bg = [make('v-col-bg'), make('v-col-bg')];
    const trans = make('v-col-trans');
    this.root.appendChild(root);
    if (this.layout === 'vertical') root.classList.add('is-vertical');
    this._verticalColumns = { root, main, bg, trans };
  }
```
